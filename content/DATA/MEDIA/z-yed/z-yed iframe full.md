        <![CDATA[
          var panZoomInstance;
          var targetZoom = 0.80;
          var targetX = 3620;
          var targetY = 360;

          function resetToInitial() {
            if (!panZoomInstance) return;

            // 1. Reset zoom level first
            panZoomInstance.zoom(targetZoom);

            // 2. Fetch viewport scale sizes safely without recalculating matrix bounding boxes
            var sizes = panZoomInstance.getSizes();
            var realZoom = sizes.realZoom;

            // 3. Compute static center offsets
            var panX = (sizes.width / 2) - (targetX * realZoom);
            var panY = (sizes.height / 2) - (targetY * realZoom);

            panZoomInstance.pan({ x: panX, y: panY });
          }

          function getFullscreenDoc() {
            return window.parent ? window.parent.document : document;
          }

          function isFullscreenActive() {
            var doc = getFullscreenDoc();
            return !!(doc.fullscreenElement || doc.webkitFullscreenElement || document.fullscreenElement || document.webkitFullscreenElement);
          }

          function updateFullscreenUI() {
            var svgRoot = document.getElementById('interactive-diagram');
            
            if (isFullscreenActive()) {
              svgRoot.classList.add('is-fullscreen');
            } else {
              svgRoot.classList.remove('is-fullscreen');
            }

            if (panZoomInstance) {
              panZoomInstance.resize();
              resetToInitial();
            }
          }

          function toggleFullScreen() {
            var doc = getFullscreenDoc();
            var iframeElem = window.frameElement || doc.documentElement;

            if (!isFullscreenActive()) {
              if (iframeElem.requestFullscreen) {
                iframeElem.requestFullscreen();
              } else if (iframeElem.webkitRequestFullscreen) {
                iframeElem.webkitRequestFullscreen();
              }
            } else {
              exitFullScreen();
            }
          }

          function exitFullScreen() {
            var doc = getFullscreenDoc();
            if (doc.exitFullscreen) {
              doc.exitFullscreen();
            } else if (doc.webkitExitFullscreen) {
              doc.webkitExitFullscreen();
            } else if (document.exitFullscreen) {
              document.exitFullscreen();
            } else if (document.webkitExitFullscreen) {
              document.webkitExitFullscreen();
            }
          }

          function moveDiagram(direction) {
            if (!panZoomInstance) return;
            var step = 200;
            if (direction === 'up') panZoomInstance.panBy({x: 0, y: step});
            if (direction === 'down') panZoomInstance.panBy({x: 0, y: -step});
            if (direction === 'left') panZoomInstance.panBy({x: step, y: 0});
            if (direction === 'right') panZoomInstance.panBy({x: -step, y: 0});
          }

          function bindControl(elementId, actionFn) {
            var elem = document.getElementById(elementId);
            if (!elem) return;
            var handleAction = function(e) {
              e.preventDefault();
              e.stopPropagation();
              actionFn();
            };
            elem.addEventListener('touchstart', handleAction, { passive: false });
            elem.addEventListener('click', handleAction);
          }

          window.addEventListener("load", function() {
            panZoomInstance = svgPanZoom('#interactive-diagram', {

              controlIconsEnabled: true,

            bindControl('btn-right', function() { moveDiagram('right'); });
